function updateClock() { const now = new Date(); const clockEl = document.getElementById('hudClock'); if (clockEl) { clockEl.textContent = now.toTimeString().split(' ')[0]; } const cpuEl = document.getElementById('cpuVal'); const ramEl = document.getElementById('ramVal'); if (cpuEl && ramEl) { const cpu = 20 + Math.floor(Math.random() * 18); const ram = (4.1 + Math.random() * 0.4).toFixed(1); cpuEl.textContent = cpu + '%'; ramEl.textContent = ram + ' GB'; } } setInterval(updateClock, 1000); updateClock();
